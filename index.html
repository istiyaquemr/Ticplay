const cells = document.querySelectorAll(".cell"); const statusText = document.getElementById("status"); let board = Array(9).fill(""); let gameActive = true; let scores = { X: 0, O: 0, draws: 0 }; const winningCombinations = [ [0,1,2], [3,4,5], [6,7,8], [0,3,6], [1,4,7], [2,5,8], [0,4,8], [2,4,6] ]; // Player X move cells.forEach(cell => { cell.addEventListener("click", () => { const index = Number(cell.dataset.index); if (!gameActive || board[index] !== "") return; // Player move makeMove(index, "X"); if (checkWinner("X")) return; if (board.every(value => value !== "")) { finishDraw(); return; } // Computer turn statusText.textContent = "🤖 Computer is thinking..."; setTimeout(computerMove, 500); }); }); // Computer randomly finds an empty cell function computerMove() { if (!gameActive) return; const emptyCells = []; board.forEach((value, index) => { if (value === "") { emptyCells.push(index); } }); if (emptyCells.length === 0) return; // Select a random empty cell const randomIndex = Math.floor( Math.random() * emptyCells.length ); const computerChoice = emptyCells[randomIndex]; // Computer plays O makeMove(computerChoice, "O"); if (checkWinner("O")) return; if (board.every(value => value !== "")) { finishDraw(); return; } statusText.textContent = "Your turn (X)"; } // Put X or O on the board function makeMove(index, player) { board[index] = player; cells[index].textContent = player; cells[index].classList.add(player.toLowerCase()); cells[index].disabled = true; } // Check winner function checkWinner(player) { const winningCombo = winningCombinations.find(combo => combo.every(index => board[index] === player) ); if (winningCombo) { statusText.textContent = "🎉 Player " + player + " Wins!"; scores[player]++; updateScores(); winningCombo.forEach(index => { cells[index].style.background = "#166534"; }); gameActive = false; cells.forEach(cell => cell.disabled = true); return true; } return false; } // Draw result function finishDraw() { statusText.textContent = "🤝 It's a Draw!"; scores.draws++; updateScores(); gameActive = false; } // Update scoreboard function updateScores() { document.getElementById("scoreX").textContent = scores.X; document.getElementById("scoreO").textContent = scores.O; document.getElementById("draws").textContent = scores.draws; } // Restart game function restartGame() { board = Array(9).fill(""); gameActive = true; cells.forEach(cell => { cell.textContent = ""; cell.disabled = false; cell.className = "cell"; cell.style.background = ""; }); statusText.textContent = "Your turn (X)"; } // Reset scores function resetScores() { scores = { X: 0, O: 0, draws: 0 }; updateScores(); restartGame(); } document.getElementById("restart") .addEventListener("click", restartGame); document.getElementById("resetScores") .addEventListener("click", resetScores); .
